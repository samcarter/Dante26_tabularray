% Was ist tabularray
\begin{frame}
  \frametitle{Was ist Tabularray}
  \begin{columns}
    \begin{column}{.65\textwidth}
      \begin{itemize}
        \item modernes Paket für Tabellen und Arrays
        \item veröffentlicht in 2021 von Jianrui Lyu
        \item adoptiert von den TeXackers
        \item erlaubt Trennung von Format und Inhalt
        \item sehr mächtig (heute Grundlagen und Highlights)
      \end{itemize}   
    \end{column}
    \begin{column}{.35\textwidth}
      \hfill\color{themecolor}\lucideicon[height=4cm]{message-circle-question-mark}\hspace*{1cm}
    \end{column}
  \end{columns}  
\end{frame}

% Spalten
\begin{frame*}
  \frametitle{Spalten}
  \begin{tcblisting}{}
    \begin{tblr}{|r|c|l|X[halign=c]|}
      text & text & text & text           \\
      a    & a    & a    & text text text \\
    \end{tblr}
  \end{tcblisting}  
  \begin{tcblisting}{}
    \begin{tblr}{
      colspec={|r|Q[wd=1cm,halign=c]|},
      column{3}={halign=l}
    }
      text & text & text\\
      a    & a    & a   \\
    \end{tblr}
  \end{tcblisting}    
\end{frame*}

% Zeilen
\begin{frame*}
  \frametitle{Zeilen}
  \begin{tcblisting}{}
    \usepackage{xcolor}
   
    \begin{tblr}{
      rowspec={Q[t]Q[m]Q[b]},
      row{1}={font=\bfseries},
      row{odd}={bg=lightgray}
    }
    {Text \\ Text} & a & a \\
    a & a & {Text \\ Text} \\
    a & {Text \\ Text} & a \\
    \end{tblr}
  \end{tcblisting}  
\end{frame*}

% Zellen
\begin{frame*}

      1 & 2 & 3\\
      4 & 5 & 6\\ 
    \end{tblr}
  \end{tcblisting}  
\end{frame*}

% Zellen verbinden
\begin{frame*}
  \frametitle{Zellen kombinieren}
  \begin{tcblisting}{}
    \begin{tblr}{
      vlines,hlines,
      hspan=even,
      cell{2}{2}={c=2}{},
    }

      & Langer Text & \\
      & 5 & 6\\ 
    \end{tblr}
  \end{tcblisting}
\end{frame*}

% zählen
\begin{frame*}
  \frametitle{Das kleine Einmaleins}
  \begin{tcblisting}{}
    \begin{tblr}{}
      x & \therownum & x \\
      \therownum & x & x \\
      \thecolcount/\therowcount & x & x \\
    \end{tblr}
  \end{tcblisting}
  \begin{tcblisting}{}
    \begin{tblr}{
      row{1}={cmd=\thecolnum},
      column{1}={cmd=\therownum},
    }
      &   &  \\
      & x & x\\
    \end{tblr}
  \end{tcblisting}  
\end{frame*}
